🚀 XLS to XLSM Batch Converter

一個高效、支援遞迴搜尋的自動化 Python 工具，專門將舊版 Excel (.xls) 批次轉換為啟用巨集的活頁簿 (.xlsm)。

✨ 主要特色

  • 遞迴資料夾掃描： 自動深入所有子資料夾，一網打盡所有需要轉換的檔案。
  • 完整保留巨集： 透過微軟官方 Excel 核心 COM 自動化（FileFormat=52），確保 VBA 程式碼與格式 100% 不失真。
  • 錯誤防護機制： 遇到毀損或被佔用的檔案會自動跳過並記錄，不會導致整批工作崩潰。
import os
import sys
import time
import win32com.client as win32

def ultimate_convert_xls_to_xlsm(root_folder):
    """
    遞迴搜尋根資料夾及所有子資料夾中的 .xls 檔案，批次轉換為 .xlsm 格式
    """
    abs_root = os.path.abspath(root_folder)
    
    if not os.path.exists(abs_root):
        print(f"❌ 找不到指定的根資料夾: {abs_root}")
        return

    print(f"🔍 開始掃描資料夾及所有子資料夾: {abs_root}")
    
    target_files = []
    for dirpath, _, filenames in os.walk(abs_root):
        for filename in filenames:
            if filename.lower().endswith(".xls") and not filename.startswith("~$"):
                full_path = os.path.join(dirpath, filename)
                target_files.append(full_path)

    total_files = len(target_files)
    if total_files == 0:
        print("⚠️ 在該資料夾與其子資料夾中找不到任何 .xls 檔案。")
        return

    print(f"🎯 總共找到 {total_files} 個 .xls 檔案準備轉換...\n" + "="*50)

    try:
        excel = win32.Dispatch("Excel.Application")
        excel.Visible = False
        excel.DisplayAlerts = False
    except Exception as e:
        print(f"❌ 無法啟動 Excel 應用程式，請確認是否正確安裝 Office 且 pywin32 正常運作。\n錯誤訊息: {e}")
        return

    success_count = 0
    fail_count = 0
    start_time = time.time()

    for idx, input_path in enumerate(target_files, 1):
        filename = os.path.basename(input_path)
        dir_name = os.path.dirname(input_path)
        
        output_filename = os.path.splitext(filename)[0] + ".xlsm"
        output_path = os.path.join(dir_name, output_filename)

        print(f"[{idx}/{total_files}] 處理中: {filename}")
        
        try:
            wb = excel.Workbooks.Open(input_path)
            wb.SaveAs(output_path, FileFormat=52)
            wb.Close(False)
            
            print(f"   └── 成功 ➔ {output_filename}")
            success_count += 1
            
        except Exception as e:
            print(f"   └── ❌ 失敗！跳過此檔案。原因: {e}")
            fail_count += 1

    try:
        excel.Quit()
    except:
        pass

    elapsed_time = time.time() - start_time
    print("="*50)
    print(f"🎉 全部作業完成！")
    print(f"⏱️ 總耗時: {elapsed_time:.2f} 秒")
    print(f"✅ 成功: {success_count} 個")
    print(f"❌ 失敗: {fail_count} 個")

if __name__ == "__main__":
    target_dir = "./"
    ultimate_convert_xls_to_xlsm(target_dir)
